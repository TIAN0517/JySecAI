# Infrastructure Secrets — secret-regex patterns
# Source: Claude-OSINT patterns, adapted for JySecAI
# 高危：私鑰/連線字串命中 = critical finding；HITL 審計必須 reject 對其的任何寫入操作

# ── 私鑰區塊 ──
PRIVATE_KEY_RSA     = -----BEGIN RSA PRIVATE KEY-----
PRIVATE_KEY_EC      = -----BEGIN EC PRIVATE KEY-----
PRIVATE_KEY_OPENSSH = -----BEGIN OPENSSH PRIVATE KEY-----
PRIVATE_KEY_PKCS8   = -----BEGIN PRIVATE KEY-----
KUBECONFIG_CERT     = (?i)client-certificate-data:\s*[A-Za-z0-9+/]{100,}

# ── 資料庫連線字串 ──
MONGO_URI           = mongodb(?:\+srv)?:\/\/[^\/\s:]+:[^\/\s@]+@[^\s\/]+
POSTGRES_URI        = postgres(?:ql)?:\/\/[^\/\s:]+:[^\/\s@]+@[^\s\/]+
MYSQL_URI           = mysql:\/\/[^\/\s:]+:[^\/\s@]+@[^\s\/]+
REDIS_URI           = redis(?:s)?:\/\/[^\s:]+:[^\s@]+@[^\s\/]+   # 已知：compose.jysec.yml 在用
DATABASE_URL_GENERIC= (?i)database[_-]?url\s*[:=]\s*['\"]?[^\s'\"]{16,}

# ── 雲儲存 ──
S3_BUCKET_URL       = [a-z0-9][a-z0-9\-]{2,62}\.s3\.amazonaws\.com
AZURE_CONN_STRING   = (?i)defaultendpointsprotocol=https;accountname=[^;]+;accountkey=[A-Za-z0-9+/=]{40,}
GCS_SERVICE_ACCOUNT = (?i)"type":\s*"service_account"

# ── .env / 配置洩露 ──
ENV_FILE_ASSIGN     = (?i)^\s*[A-Z][A-Z0-9_]{2,}\s*=\s*['\"]?[^\s'\"]{8,}['\"]?\s*$
DOTENV_SECRET_KEY   = (?i)(secret|password|token|api[_-]?key)\s*[:=]

# 落盤：recon/<target-slug>/secrets-findings.md
