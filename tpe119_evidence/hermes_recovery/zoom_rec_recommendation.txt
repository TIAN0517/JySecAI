"""Multi-source scanner: ZoomEye + FullHunter + Shodan for EPCR endpoints."""
import requests, json, time, sys, argparse
from concurrent.futures import ThreadPoolExecutor

API_KEY="0F4f37...16E7"
ZOOMEYE_API = "https://api.zo.me/v1/search/unified"

def zoomeye_search(query, page=1):
    headers = {"API-Key": API_KEY}
    params = {"keywords": query.replace(" ", "%20"), "page": page, "num": 20}
    try:
        resp = requests.get(ZOOMEYE_API, headers=headers, params=params, timeout=30)
        if resp.status_code == 200:
            data = resp.json()
            return data.get("list", [])
    except Exception as e:
        print(f"[-] ZoomEye error: {e}")
    return []

def parse_results(results):
    assets = []
    for item in results:
        ip = item.get("ip", "")
        ports = item.get("port", 443)
        product = item.get("product", "")
        certs = json.loads(item.get("cert", "{}")) if item.get("cert") else {}
        subject = certs.get("subject", {}) if isinstance(certs, dict) else {}
        domains = subject.get("DNS", [])
        for d in domains:
            if "epcr" in d.lower() or "gov" in d.lower():
                assets.append({"host": d, "ip": ip, "port": ports, "product": product})
    return assets

def main(domains):
    queries = [
        f'app="EPCR" country:"TW" {d}' if "epcr" in d.lower() else f'product:"nginx" country:"TW" title:"EPCR" {d}'

    all_assets = []
    with ThreadPoolExecutor(max_workers=3) as executor:
        futures = [executor.submit(zoomeye_search, q.strip(), p) for q in queries for p in range(1,4)]
        for future in futures:
            results = future.result()
            assets = parse_results(results)
            all_assets.extend(assets)
            if assets:
                for a in assets:
                    print(json.dumps(a))
    unique_hosts = list({a["host"] for a in all_assets})

    print(f"\n[+] Total unique EPCR-like hosts: {len(unique_hosts)}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--zoomeye-key", default=API_KEY)
