#!/usr/bin/env python3
"""ZoomEye EPCR Scanner using API-Key + Points."""
import requests, json, time, sys, os
from concurrent.futures import ThreadPoolExecutor

API_KEY="0F4f37...16E7"
HEADERS = {"API-Key": API_KEY}
BASE_URL = "https://api.zo.me/v1/search/unified"

QUERIES = [
    'app="EPCR" country:"TW"',
    'product:"Apache" country:"TW" title:"login"',
    'port:4004 country:"TW" title:"Auth"',
    'product:"nginx" country:"TW" after:"2024-01-01"',
]

def zoomeye_search(query, page=1, num=20):
    """Single ZoomEye API request."""
    params = {"keywords": query.replace(" ", "%20"), "page": page, "num": num}
    try:
        resp = requests.get(BASE_URL, headers=HEADERS, params=params, timeout=30)
        if resp.status_code == 401:
            raise ValueError("API-Key 無效或 Points 不足")
        elif resp.status_code != 200:
            return []
        data = resp.json()
        if isinstance(data, dict) and data.get("error"):
            raise ValueError(data["error"])
        return data.get("list", [])
    except Exception as e:
        print(f"[-] {query} -> {e}")
        return []

def extract_assets(results):
    """Parse ZoomEye results into structured asset list."""
    assets = []
    for item in results:
        ip = item.get("ip", "")
        port = item.get("port", "")
        product = item.get("product", "")
        certs = json.loads(item.get("cert", "{}")) if item.get("cert") else {}
        subject = certs.get("subject", {}) if isinstance(certs, dict) else {}
        domains = subject.get("DNS", []) if isinstance(subject, dict) else []

        for d in domains:
            assets.append({
                "host": d.rstrip("."),
                "ip": ip,
                "port": port,
                "product": product,
                "ssl": bool(certs)
            })
    return assets

def main():
    print("[*] Starting ZoomEye EPCR Scan...")
    all_assets = []

    with ThreadPoolExecutor(max_workers=3) as executor:
        futures = {}
        for q in QUERIES:
            for page in range(1, 4):  # 每個查詢跑 3 頁
                future = executor.submit(zoomeye_search, q, page)
                futures[future] = q

        for future in futures:
            query = futures[future]
            results = future.result()
            assets = extract_assets(results)
            all_assets.extend(assets)
            print(f"[+] {query} Page {futures[future]} -> {len(assets)} assets")

    # 去重並儲存
    seen = set()
    unique_assets = []
    for a in all_assets:
        key = a["host"]
        if key not in seen:
            seen.add(key)
            unique_assets.append(a)

    output_path = "tpe119_evidence/zoomeye_results/epcr_taiwan_gov_assets.json"
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(unique_assets, f, ensure_ascii=False, indent=2)

    print(f"\n[✔] 總共發現 {len(unique_assets)} 個唯一 EPCR 相關資產")
    print(f"[✔] 結果已儲存至 {output_path}")

if __name__ == "__main__":
    main()
