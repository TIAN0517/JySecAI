<?php
class Router {
	private static $routes = [
		"^\/([^*]+)$",
		"^\/([^*]+)\/?$",
		"^\/([^*]+)\/([^*]+)\/?$",
		"^\/([^*]+)\/([^*]+)\/([^*]+)\/?$",
		"^\/([^*]+)\/([^*]+)\/([^*]+)\/([^*]+)\/?$",
		"^\/([^*]+)\/([^*]+)\/([^*]+)\/([^*]+)\/([^*]+)\/?$"
	];
	private static $parameters = [];
	public function __construct() {
		$folderName = str_replace("/index.php", "", $_SERVER['PHP_SELF']);
		$redirectUrl = ( isset($_SERVER['REDIRECT_URL']) )? $_SERVER['REDIRECT_URL']:explode("?", $_SERVER['REQUEST_URI'] )[0];
		if($folderName != '') $url = '/' . str_replace($folderName . '/', "", $redirectUrl);
		else $url = str_replace($folderName, "", $redirectUrl);
		foreach (self::$routes as $route) {
			if (!preg_match("/" . $route . "/", $url, $matches))
				continue;
			self::$parameters = array_slice($matches, 1);
		}
	}
	public static function getParameter($index){
		if(isset(self::$parameters[($index-1)])) return str_replace('/', '', self::$parameters[($index-1)]);
		else return "";
	}
}
