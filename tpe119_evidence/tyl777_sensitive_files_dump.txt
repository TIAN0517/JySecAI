<?php
error_reporting(E_ALL ^ E_NOTICE);

$webSocket = new SocketService('0.0.0.0', 9453);

class SocketService{
	public $sockets;		// socket連接池
	public $users;			// 所有client连接进来的信息，包括socket、client名字等
	public $master;			// socket resource
	
	// 開始Socket連線
	public function __construct($address, $port) {
		// 開始Socket連線並監聽
		$server = socket_create(AF_INET, SOCK_STREAM, SOL_TCP);
		socket_set_option($server, SOL_SOCKET, SO_REUSEADDR, 1);
		socket_bind($server, $address, $port);
		socket_listen($server);
		$this->master = $server;	// 主機資源名稱

		// 將主機資源名稱塞入資源池
		$this->sockets = [$this->master];
	}
	
	//对创建的socket循环进行监听，处理数据	
	function running(){
		// 無限循環至Sokcet斷開
		while(true){
			// sendLine("running... " . socket_strerror(socket_last_error($this->master)) );
			
			// 判斷監聽資料是否有變化
			$changes = $this->sockets;
			$clientWrite = null;	// 客戶端是否有資料；null = 不管是否有改寫變化
			$except = null;			// 要排除的資料；null = 全部監聽
			$timeOut = null;		// 超時秒數；null = 有新動作retern
			socket_select($changes, $write, $except, $timeOut);
			
			foreach($changes as $sock){
				// 連接池沒改變表示為新用戶
				if($sock == $this->master){
					// 將新的連線(client)綁定到主機資源中
					$client = socket_accept($this->master);

					// 將新的連線建立唯一值
					$key = uniqid();
					$this->sockets[] = $client;	// 將新的連線加入資源池中
					$this->users[$key] = [ 'socket' => $client, 'shou' => false];	// 建立新的使用者，shou => 三向交握狀態
				} else {
					$recvLen = 0;
					$buffer = '';
					// 讀取Socket內容，每次讀取1000bit，
					do {
						$l = socket_recv($sock, $buf, 1000, 0);
						$recvLen += $l;
						$buffer .= $buf;
					} while ($l == 1000);

					// 找出該連線對應的ID
					$clientKey = false;
					foreach ($this->users as $k => $v) {
						if($sock == $v['socket']) $clientKey = $k;
					}

					// 長度小於7 或 找不到使用者表示連線已斷開
					if($recvLen < 7 || $clientKey == false) {
						$this->close($clientKey);
						continue;
					}

					// 坢斷該連線是否已三向交握
					if(!$this->users[$clientKey]['shou']) {
						$this->woshou($clientKey, $buffer);
					} else {
						// 針對收到的訊息進行解碼處理
						$buffer = $this->uncode($buffer, $clientKey);
						if($buffer == false){
							continue;
						}
						// 確定有資訊，將訊息進行處理
						$this->msgProcessing($buffer);
					}
				}
			}
			
		}
		
	}
	
	// 關閉特定連線
	function close($clientKey){
		// 斷開對應Socket
		socket_close($this->users[$clientKey]['socket']);
		// 刪除連線列表內容
		unset($this->users[$clientKey]);

		$this->sockets= [$this->master];

		// 重新整理資源池
		foreach($this->users as $v){
			$this->sockets[] = $v['socket'];
		}
	}
	
	// 對client開始進行三向交握
	function woshou($clientKey, $buffer){

		// 取Sec-WebSocket-Key的值加密；$key + 258EAFA5-E914-47DA-95CA-C5AB0DC85B11神秘字串
		$buf  = substr($buffer,strpos($buffer,'Sec-WebSocket-Key:') + 18);
		$key  = trim(substr($buf, 0, strpos($buf, "\r\n")));
		$returnKey = base64_encode(sha1($key . "258EAFA5-E914-47DA-95CA-C5AB0DC85B11", true));
		
		$new_message = "HTTP/1.1 101 Switching Protocols\r\n";
		$new_message .= "Upgrade: websocket\r\n";
		$new_message .= "Sec-WebSocket-Version: 13\r\n";
		$new_message .= "Connection: Upgrade\r\n";
		$new_message .= "Sec-WebSocket-Accept: " . $returnKey . "\r\n\r\n";
		socket_write($this->users[$clientKey]['socket'],$new_message,strlen($new_message));
		$this->users[$clientKey]['shou'] = true;
		return true;
		
	}
	
	// 解碼訊息
	function uncode($buffer, $clientKey){
		$mask = [];
		$data = '';
		$bufferPack = unpack('H*', $buffer);	// 將原本二進制解碼為十六進制
		$head = substr($bufferPack[1], 0, 2);
		if ($head == '81' && !isset($this->slen[$clientKey])) {
			$len = substr($bufferPack[1], 2, 2);
			$len = hexdec($len);
			if(substr($bufferPack[1] , 2, 2) == 'fe') {
				$len = substr($bufferPack[1], 4, 4);
				$len = hexdec($len);
				$bufferPack[1] = substr($bufferPack[1], 4);
			} else if(substr($bufferPack[1], 2, 2) == 'ff') {
				$len = substr($bufferPack[1], 4, 16);
				$len = hexdec($len);
				$bufferPack[1] = substr($bufferPack[1], 16);
			}
			$mask[] = hexdec(substr($bufferPack[1], 4, 2));
			$mask[] = hexdec(substr($bufferPack[1], 6, 2));
			$mask[] = hexdec(substr($bufferPack[1], 8, 2));
			$mask[] = hexdec(substr($bufferPack[1], 10, 2));
			$s = 12;
			$n = 0;
		} else if($this->slen[$clientKey] > 0) {
			$len = $this->slen[$clientKey];
			$mask = $this->ar[$clientKey];
			$n = $this->n[$clientKey];
			$s = 0;
		}
		
		$e = strlen($bufferPack[1])-2;
		for ($i = $s; $i <= $e; $i += 2) {
			$data .= chr($mask[$n%4]^hexdec(substr($bufferPack[1], $i, 2)));
			$n++;
		}
		$dlen = strlen($data);
		
		if($len > 255 && $len > $dlen + intval($this->sjen[$clientKey])) {
			$this->ar[$clientKey] = $mask;
			$this->slen[$clientKey] = $len;
			$this->sjen[$clientKey] = $dlen + intval($this->sjen[$clientKey]);
			$this->sda[$clientKey] = $this->sda[$clientKey] . $data;
			$this->n[$clientKey] = $n;
			return false;
		} else {
			unset($this->ar[$clientKey], $this->slen[$clientKey], $this->sjen[$clientKey], $this->n[$clientKey]);
			$data = $this->sda[$clientKey] . $data;
			unset($this->sda[$clientKey]);
			return $data;
		}
		
	}
	
	// 對傳送得資訊進行編碼
	function code($msg){
		$frame = [];
		$frame[0] = '81';
		$len = strlen($msg);
		if($len < 126) {
			$frame[1] = $len < 16? '0' . dechex($len):dechex($len);
		} else if($len < 65025) {
			$s  =dechex($len);
			$frame[1] = '7e' . str_repeat('0', 4 - strlen($s)) . $s;
		} else {
			$s = dechex($len);
			$frame[1] = '7f' . str_repeat('0', 16 - strlen($s)) . $s;
		}
		// 將訊息轉換為十六進制
		$msgHex = '';
		$msgLen = strlen($msg);
		for ($i = 0; $i < $msgLen; $i++) {
			$msgHex .= dechex(ord($msg{$i}));
		}
		$frame[2] = $msgHex;
		$data = implode('', $frame);
		return pack("H*", $data);
	}

	function msgProcessing($buffer) {
		$msg = json_decode($buffer, true);
		if( isset($msg['event']) ) {
			switch ($msg['event']) {
				case 'newWish':
					$msg['event'] = 'newWish';
					$msg['time'] = date("Y-m-d H:i:s");
					$this->sendClient($msg);
					break;
				
				default:
					$msg['online'] = count($this->users);
					$msg['time'] = date("Y-m-d H:i:s");
					$this->sendClient($msg);
			}
		}
	}

	function sendClient($msg) {
		$users = $this->users;
		$msg = $this->code(json_encode($msg));
		foreach($users as $user){
			socket_write($user['socket'], $msg, strlen($msg));
		}
	}
}

function curlPost($url = '', $data = [], $header = [] ){
	$ch = curl_init();
	curl_setopt_array($ch, array(
		CURLOPT_URL => $url,
		CURLOPT_POST => TRUE,
		CURLOPT_RETURNTRANSFER => TRUE,
		CURLOPT_SSL_VERIFYHOST => 1,
		CURLOPT_SSL_VERIFYPEER => 0,
		CURLOPT_HTTPHEADER => array(implode(' ',$header)),
		CURLOPT_POSTFIELDS => http_build_query($data),
		CURLOPT_TIMEOUT => 5000,
	));
	$result = curl_exec($ch);
	curl_close($ch);
	return $result;
}
