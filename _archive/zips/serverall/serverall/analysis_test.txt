        <script>
        function analyzePayload() {
            const payload = document.getElementById('payloadInput').value;
            if (!payload.trim()) {
                alert('请输入XSS载荷');
                return;
            }
            
            // 模拟分析逻辑
            const result = document.getElementById('result');
            let riskLevel = 'low';
            let techniques = [];
            
            // 基础分析
            if (payload.includes('<script>')) {
                techniques.push('脚本标签注入');
                riskLevel = 'high';
            }
            if (payload.includes('on')) {
                techniques.push('事件处理器注入');
                riskLevel = 'high';
            }
            if (payload.includes('javascript:')) {
                techniques.push('JavaScript协议');
                riskLevel = 'high';
            }
            if (payload.includes('<img')) {
                techniques.push('图片标签注入');
                riskLevel = 'medium';
            }
            if (payload.includes('<svg')) {
                techniques.push('SVG注入');
                riskLevel = 'medium';
            }
            
            let riskClass = riskLevel + '-risk';
            let riskColor = riskLevel === 'high' ? '#ff0000' : riskLevel === 'medium' ? '#ffa500' : '#00ff41';
            
            result.innerHTML = `
                <h3 style="color: ${riskColor}">风险等级: ${riskLevel.toUpperCase()}</h3>
                <p><strong>载荷长度:</strong> ${payload.length} 字符</p>
                <p><strong>检测到的技术:</strong> ${techniques.join(', ') || '无'}</p>
                <p><strong>载荷内容:</strong></p>
                <pre style="background: rgba(0,0,0,0.5); padding: 10px; overflow-x: auto;">${payload}</pre>
            `;
            result.className = 'result ' + riskClass;
        }
        </script>
